using System;
using System.Data;
using System.Windows.Forms;
using Microsoft.Data.SqlClient;

namespace harmacyManagementSystem
{
    public partial class Screen3 : Form
    {
        public Screen3()
        {
            InitializeComponent();
            LoadMedicines();
        }

        // ================= LOAD DATA =================
        private void LoadMedicines()
        {
            using (SqlConnection con = DBConnection.GetConnection())
            {
                con.Open();
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Medicines", con);
                DataTable dt = new DataTable();
                da.Fill(dt);
                dataGridView1.DataSource = dt;
            }
        }

        // ================= VALIDATION =================
        private bool ValidateInputs()
        {
            if (txtMedicineName.Text == "" ||
                txtCategory.Text == "" ||
                txtCompany.Text == "" ||
                txtPrice.Text == "" ||
                txtQuantity.Text == "" ||
                txtExpiryDate.Text == "")
            {
                MessageBox.Show("Fill all fields!");
                return false;
            }

            if (!decimal.TryParse(txtPrice.Text, out _))
            {
                MessageBox.Show("Invalid Price!");
                return false;
            }

            if (!int.TryParse(txtQuantity.Text, out _))
            {
                MessageBox.Show("Invalid Quantity!");
                return false;
            }

            if (!DateTime.TryParse(txtExpiryDate.Text, out _))
            {
                MessageBox.Show("Invalid Date!");
                return false;
            }

            return true;
        }

        // ================= ADD =================
        private void btnAdd_Click(object sender, EventArgs e)
        {
            if (!ValidateInputs()) return;

            using (SqlConnection con = DBConnection.GetConnection())
            {
                con.Open();

                string query = @"INSERT INTO Medicines 
                (MedicineName, Category, Company, Price, Quantity, ExpiryDate) 
                VALUES (@name,@cat,@comp,@price,@qty,@exp)";

                SqlCommand cmd = new SqlCommand(query, con);
                cmd.Parameters.AddWithValue("@name", txtMedicineName.Text);
                cmd.Parameters.AddWithValue("@cat", txtCategory.Text);
                cmd.Parameters.AddWithValue("@comp", txtCompany.Text);
                cmd.Parameters.AddWithValue("@price", Convert.ToDecimal(txtPrice.Text));
                cmd.Parameters.AddWithValue("@qty", Convert.ToInt32(txtQuantity.Text));
                cmd.Parameters.AddWithValue("@exp", Convert.ToDateTime(txtExpiryDate.Text));

                cmd.ExecuteNonQuery();
                MessageBox.Show("Added Successfully");

                LoadMedicines();
                ClearFields();
            }
        }

        // ================= UPDATE =================
        private void btnUpdate_Click(object sender, EventArgs e)
        {
            if (txtMedicineID.Text == "")
            {
                MessageBox.Show("Select record first!");
                return;
            }

            if (!ValidateInputs()) return;

            using (SqlConnection con = DBConnection.GetConnection())
            {
                con.Open();

                string query = @"UPDATE Medicines SET 
                MedicineName=@name, Category=@cat, Company=@comp,
                Price=@price, Quantity=@qty, ExpiryDate=@exp
                WHERE MedicineID=@id";

                SqlCommand cmd = new SqlCommand(query, con);
                cmd.Parameters.AddWithValue("@id", Convert.ToInt32(txtMedicineID.Text));
                cmd.Parameters.AddWithValue("@name", txtMedicineName.Text);
                cmd.Parameters.AddWithValue("@cat", txtCategory.Text);
                cmd.Parameters.AddWithValue("@comp", txtCompany.Text);
                cmd.Parameters.AddWithValue("@price", Convert.ToDecimal(txtPrice.Text));
                cmd.Parameters.AddWithValue("@qty", Convert.ToInt32(txtQuantity.Text));
                cmd.Parameters.AddWithValue("@exp", Convert.ToDateTime(txtExpiryDate.Text));

                cmd.ExecuteNonQuery();
                MessageBox.Show("Updated Successfully");

                LoadMedicines();
                ClearFields();
            }
        }

        // ================= DELETE =================
        private void btnDelete_Click(object sender, EventArgs e)
        {
            if (txtMedicineID.Text == "")
            {
                MessageBox.Show("Select record first!");
                return;
            }

            using (SqlConnection con = DBConnection.GetConnection())
            {
                con.Open();

                string query = "DELETE FROM Medicines WHERE MedicineID=@id";

                SqlCommand cmd = new SqlCommand(query, con);
                cmd.Parameters.AddWithValue("@id", Convert.ToInt32(txtMedicineID.Text));

                cmd.ExecuteNonQuery();
                MessageBox.Show("Deleted Successfully");

                LoadMedicines();
                ClearFields();
            }
        }

        // ================= CLEAR =================
        private void btnClear_Click(object sender, EventArgs e)
        {
            ClearFields();
        }

        private void ClearFields()
        {
            txtMedicineID.Clear();
            txtMedicineName.Clear();
            txtCategory.Clear();
            txtCompany.Clear();
            txtPrice.Clear();
            txtQuantity.Clear();
            txtExpiryDate.Clear();
        }

        // ================= GRID CLICK =================
        private void dataGridView1_CellClick(object sender, DataGridViewCellEventArgs e)
        {
            if (e.RowIndex >= 0)
            {
                var row = dataGridView1.Rows[e.RowIndex];

                txtMedicineID.Text = row.Cells["MedicineID"].Value?.ToString();
                txtMedicineName.Text = row.Cells["MedicineName"].Value?.ToString();
                txtCategory.Text = row.Cells["Category"].Value?.ToString();
                txtCompany.Text = row.Cells["Company"].Value?.ToString();
                txtPrice.Text = row.Cells["Price"].Value?.ToString();
                txtQuantity.Text = row.Cells["Quantity"].Value?.ToString();
                txtExpiryDate.Text = row.Cells["ExpiryDate"].Value?.ToString();
            }
        }

        // 🔥 DESIGNER FIX (IMPORTANT)
        private void btnAdd_Click_1(object sender, EventArgs e) { btnAdd_Click(sender, e); }
        private void btnUpdate_Click_1(object sender, EventArgs e) { btnUpdate_Click(sender, e); }
        private void btnDelete_Click_1(object sender, EventArgs e) { btnDelete_Click(sender, e); }
        private void btnClear_Click_1(object sender, EventArgs e) { btnClear_Click(sender, e); }
        private void txtMedicineID_TextChanged(object sender, EventArgs e) { }
    }
