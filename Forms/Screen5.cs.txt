using System;
using System.Windows.Forms;

namespace harmacyManagementSystem
{
    public partial class Screen5 : Form
    {
        decimal grandTotal = 0;

        public Screen5()
        {
            InitializeComponent();

            // 🔥 EVENT BINDING FIX
            btnAddToBill.Click += btnAddToBill_Click;
            btnClear.Click += btnClear_Click;
            btnGenerateInvoice.Click += btnGenerateInvoice_Click;
        }

        private void btnAddToBill_Click(object sender, EventArgs e)
        {
            if (txtCustomerName.Text == "" || txtMedicineName.Text == "" ||
                txtQuantity.Text == "" || txtPrice.Text == "")
            {
                MessageBox.Show("Please fill required fields!");
                return;
            }

            if (!int.TryParse(txtQuantity.Text, out int qty))
            {
                MessageBox.Show("Invalid Quantity!");
                return;
            }

            if (!decimal.TryParse(txtPrice.Text, out decimal price))
            {
                MessageBox.Show("Invalid Price!");
                return;
            }

            decimal total = qty * price;
            grandTotal += total;

            dataGridView1.Rows.Add(
                txtBillDate.Text,
                txtCustomerName.Text,
                txtPrescribedBy.Text,
                txtMedicineName.Text,
                txtBatchNo.Text,
                txtExpiryDate.Text,
                qty,
                price,
                total
            );

            txtTotal.Text = grandTotal.ToString("0.00");

            txtMedicineName.Clear();
            txtBatchNo.Clear();
            txtExpiryDate.Clear();
            txtQuantity.Clear();
            txtPrice.Clear();
        }

        private void btnClear_Click(object sender, EventArgs e)
        {
            txtBillDate.Clear();
            txtCustomerName.Clear();
            txtPrescribedBy.Clear();
            txtMedicineName.Clear();
            txtBatchNo.Clear();
            txtExpiryDate.Clear();
            txtQuantity.Clear();
            txtPrice.Clear();
            txtTotal.Clear();
            dataGridView1.Rows.Clear();
            grandTotal = 0;
        }

        private void btnGenerateInvoice_Click(object sender, EventArgs e)
        {
            if (dataGridView1.Rows.Count <= 1)
            {
                MessageBox.Show("No bill items added!");
                return;
            }

            MessageBox.Show("Invoice Generated Successfully!\nTotal Amount: " + txtTotal.Text);
        }

        // backup handlers (designer)
        private void btnAddToBill_Click_1(object sender, EventArgs e)
        {
            btnAddToBill_Click(sender, e);
        }

        private void btnClear_Click_1(object sender, EventArgs e)
        {
            btnClear_Click(sender, e);
        }

        private void btnGenerateInvoice_Click_1(object sender, EventArgs e)
        {
            btnGenerateInvoice_Click(sender, e);
        }
    }
}
